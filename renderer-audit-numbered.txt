     8	export async function render_text(

    17	    ctx.property as typeof ctx.property & {
    18	      tagline?: string | null;
    19	      classification?: string | null;
    20	      pemClassification?: string | null;
    21	      brokerProfile?: {
    22	        phone?: string | null;
    23	      } | null;
    24	    };
    25	
    26	  const values = {
    27	    classification: cleanText(
    28	      property.pemClassification ??
    29	      property.classification,
    30	    ),
    31	    title: cleanText(property.title),
    32	    subtitle: cleanText(property.tagline),
    33	    collection: cleanText(property.zoneLabel),
    34	    price: cleanText(property.price),
    35	    bedrooms:
    36	      property.bedrooms != null
    37	        ? String(property.bedrooms)
    38	        : "",
    39	    bathrooms:
    40	      property.bathrooms != null
    41	        ? (
    42	            property.halfBathrooms
    43	              ? `${property.bathrooms + 0.5}`
    44	              : `${property.bathrooms}`
    45	          )
    46	        : "",
    47	    area:
    48	      property.areaInterior != null
    49	        ? `${property.areaInterior} m²`
    50	        : "",
    51	    luxuryScore:
    52	      property.luxuryScore != null
    53	        ? String(property.luxuryScore)
    54	        : "",
    55	    phone: cleanText(
    56	      property.brokerProfile?.phone,
    57	    ),
    58	    editorial_quote:
    59	      "Permítanos asesorarle en la adquisición de esta propiedad única",
    60	  };
    61	
    62	  const fields = [
    63	    ["classification","trajanSemibold",rgb(197/255,141/255,66/255),9.65],
    64	    ["title","trajanLight",rgb(1,1,1),15],
    65	    ["subtitle","trajanLight",rgb(1,1,1),11],

     7	    .replace(/\s+/g, " ")

     9	}
    10	
    11	export function toPdfBounds(
    12	  bounds: Bounds,
    13	  artboard: Artboard,
    14	) {
    15	  return {
    16	    x: bounds.left - artboard.left,
    17	    y: bounds.bottom - artboard.bottom,
    18	    width: bounds.width,
    19	    height: bounds.height,
    20	  };
    21	}
    22	
    23	export function drawCenteredText(
    24	  page: PDFPage,
    25	  item: TemplateItem,
    26	  artboard: Artboard,
    27	  text: string,
    28	  font: PDFFont,
    29	  preferredSize: number,
    30	  color: ReturnType<typeof rgb>,
    31	) {
    32	  const box = toPdfBounds(item.bounds, artboard);
    33	
    34	  const size = preferredSize;
    35	
    36	  const textWidth = font.widthOfTextAtSize(
    37	    text,
    38	    size,
    39	  );
    40	
    41	  page.drawRectangle({
    42	    x: box.x - 3,
    43	    y: box.y - 3,
    44	    width: box.width + 6,
    45	    height: box.height + 6,
    46	    color: rgb(0, 0, 0),
    47	  });
    48	
    49	  page.drawText(text, {
    50	    x: box.x + (box.width - textWidth) / 2,
    51	    y: box.y,
    52	    size,
    53	    font,
    54	    color,
    55	  });
    56	}

    25	  value: unknown,
    26	): AssignmentMap {
    27	  if (

    33	  }

    35	  const result: AssignmentMap = {};
    36	
    37	  for (const [k, v] of Object.entries(value)) {
